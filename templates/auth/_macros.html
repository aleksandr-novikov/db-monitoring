{# Reusable form-field macro for auth pages. Wires WCAG-friendly error association: `aria-invalid="true"` and `aria-describedby="-errors"` get applied only when the field has errors, and every error in the list is rendered (not just `[0]`). `extra_class` lets the caller swap the input shape (the checkbox variant in login.html uses a different palette). #} {% macro render_field(field, input_class="") %}
{% set has_errors = field.errors|length > 0 %} {{ field( class=input_class, **({"aria-invalid": "true", "aria-describedby": field.id ~ "-errors"} if has_errors else {}) ) }} {% if has_errors %}
    {% for err in field.errors %}
  • {{ err }}
    • {% endfor %}
{% endif %}
{% endmacro %}
